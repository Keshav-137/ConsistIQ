/// Glass design tokens that change with dark/light mode.
@immutable
class GlassColors extends ThemeExtension<GlassColors> {
  final Color tintTop;
  final Color tintBottom;
  final Color border;
  final Color text;
  final Color subText;
  final List<Color> background; // 3 colors

  const GlassColors({
    required this.tintTop,
    required this.tintBottom,
    required this.border,
    required this.text,
    required this.subText,
    required this.background,
  });

  static const dark = GlassColors(
    tintTop: Color(0x2EFFFFFF),
    tintBottom: Color(0x0AFFFFFF),
    border: Color(0x4DFFFFFF),
    text: Colors.white,
    subText: Color(0xA6FFFFFF),
    background: [Color(0xFF050B1F), Color(0xFF0B1B4D), Color(0xFF0A0F2C)],
  );

  static const light = GlassColors(
    tintTop: Color(0xA6FFFFFF),
    tintBottom: Color(0x40FFFFFF),
    border: Color(0xCCFFFFFF),
    text: Color(0xFF0B1B4D),
    subText: Color(0xA60B1B4D),
    background: [Color(0xFFEAF2FF), Color(0xFFD6E6FF), Color(0xFFF1EAFF)],
  );

  @override
  GlassColors copyWith({
    Color? tintTop,
    Color? tintBottom,
    Color? border,
    Color? text,
    Color? subText,
    List<Color>? background,
  }) =>
      GlassColors(
        tintTop: tintTop ?? this.tintTop,
        tintBottom: tintBottom ?? this.tintBottom,
        border: border ?? this.border,
        text: text ?? this.text,
        subText: subText ?? this.subText,
        background: background ?? this.background,
      );

  @override
  GlassColors lerp(ThemeExtension<GlassColors>? other, double t) {
    if (other is! GlassColors) return this;
    return GlassColors(
      tintTop: Color.lerp(tintTop, other.tintTop, t)!,
      tintBottom: Color.lerp(tintBottom, other.tintBottom, t)!,
      border: Color.lerp(border, other.border, t)!,
      text: Color.lerp(text, other.text, t)!,
      subText: Color.lerp(subText, other.subText, t)!,
      background: List.generate(
          3, (i) => Color.lerp(background[i], other.background[i], t)!),
    );
  }
}

String fmtDuration(Duration d) {
  if (d.isNegative) return '0s';
  final h = d.inHours;
  final m = d.inMinutes % 60;
  final s = d.inSeconds % 60;
  if (h > 0) return '${h}h ${m}m';
  return '${m}m ${s}s';
}

/// Animated gradient + drifting glow blobs. Wrap every screen with this
/// so the whole app shares the same glassy, transparent look.
class GlassBackground extends StatefulWidget {
  final Widget child;
  const GlassBackground({super.key, required this.child});

  @override
  State<GlassBackground> createState() => _GlassBackgroundState();
}

class _GlassBackgroundState extends State<GlassBackground>
    with SingleTickerProviderStateMixin {
  late final AnimationController _c;

  @override
  void initState() {
    super.initState();
    _c = AnimationController(vsync: this, duration: const Duration(seconds: 10))
      ..repeat();
  }

  @override
  void dispose() {
    _c.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    final g = context.glass;
    final dark = context.isDark;
    final size = MediaQuery.of(context).size;

    return AnnotatedRegion<SystemUiOverlayStyle>(
      value: dark ? SystemUiOverlayStyle.light : SystemUiOverlayStyle.dark,
      child: Stack(
        fit: StackFit.expand,
        children: [
          DecoratedBox(
            decoration: BoxDecoration(
              gradient: LinearGradient(
                begin: Alignment.topLeft,
                end: Alignment.bottomRight,
                colors: g.background,
              ),
            ),
          ),
          AnimatedBuilder(
            animation: _c,
            builder: (context, child) {
              final t = _c.value * 2 * math.pi;
              return Stack(children: [
                _blob(
                    size.width * 0.15 + math.sin(t) * 30,
                    size.height * 0.12 + math.cos(t) * 30,
                    260,
                    AppTheme.blue,
                    dark),
                _blob(
                    size.width * 0.55 + math.cos(t) * 40,
                    size.height * 0.55 + math.sin(t) * 40,
                    300,
                    AppTheme.purple,
                    dark),
                _blob(
                    size.width * -0.1 + math.sin(t + 2) * 30,
                    size.height * 0.7 + math.cos(t + 2) * 30,
                    220,
                    AppTheme.cyan,
                    dark),
              ]);
            },
          ),
          widget.child,
        ],
      ),
    );
  }

  Widget _blob(double x, double y, double d, Color c, bool dark) {
    return Positioned(
      left: x,
      top: y,
      child: Container(
        width: d,
        height: d,
        decoration: BoxDecoration(
          shape: BoxShape.circle,
          gradient: RadialGradient(colors: [
            c.withValues(alpha: dark ? 0.45 : 0.35),
            c.withValues(alpha: 0.0),
          ]),
        ),
      ),
    );
  }
}

import 'dart:ui';
import 'package:flutter/material.dart';

import '../theme/app_theme.dart';

/// Reusable frosted-glass card. Use this for every card, dialog and sheet.
class GlassContainer extends StatelessWidget {
  final Widget? child;
  final double? width;
  final double? height;
  final double radius;
  final double blur;
  final EdgeInsetsGeometry? padding;
  final Color? glowColor;
  final double glowOpacity;

  const GlassContainer({
    super.key,
    this.child,
    this.width,
    this.height,
    this.radius = 28,
    this.blur = 20,
    this.padding,
    this.glowColor,
    this.glowOpacity = 0,
  });

  @override
  Widget build(BuildContext context) {
    final g = context.glass;
    final content = Container(
      width: width,
      height: height,
      padding: padding,
      decoration: BoxDecoration(
        borderRadius: BorderRadius.circular(radius),
        gradient: LinearGradient(
          begin: Alignment.topLeft,
          end: Alignment.bottomRight,
          colors: [g.tintTop, g.tintBottom],
        ),
        border: Border.all(color: g.border, width: 1.2),
      ),
      child: child,
    );
    return Container(
      decoration: BoxDecoration(
        borderRadius: BorderRadius.circular(radius),
        boxShadow: glowColor == null
            ? null
            : [
                BoxShadow(
                  color: glowColor!.withValues(alpha: glowOpacity),
                  blurRadius: 60,
                  spreadRadius: 6,
                ),
              ],
      ),
      child: ClipRRect(
        borderRadius: BorderRadius.circular(radius),
        child: blur <= 0
            ? content
            : BackdropFilter(
                filter: ImageFilter.blur(sigmaX: blur, sigmaY: blur),
                child: content,
              ),
      ),
    );
  }
}

class StudyPlan {
  String id;
  String title;
  String taskId; // linked daily task that carries the streak + reminders
  String source;
  int createdAt;
  int hour;
  int minute;
  List<StudyDay> days;

  StudyPlan({
    required this.id,
    required this.title,
    required this.taskId,
    required this.source,
    required this.createdAt,
    required this.hour,
    required this.minute,
    required this.days,
  });

  int get totalPoints => days.fold(0, (a, d) => a + d.points.length);
  int get donePoints => days.fold(0, (a, d) => a + d.doneCount);
  int get doneDays => days.where((d) => d.done).length;
  double get progress => totalPoints == 0 ? 0 : donePoints / totalPoints;

  StudyDay? get nextDay {
    for (final d in days) {
      if (!d.done) return d;
    }
    return null;
  }

  void markNextDone() {
    final d = nextDay;
    if (d == null) return;
    for (final p in d.points) {
      p.done = true;
    }
  }

  Map<String, dynamic> toJson() => {
        'id': id,
        'title': title,
        'task': taskId,
        'source': source,
        'created': createdAt,
        'h': hour,
        'm': minute,
        'days': days.map((d) => d.toJson()).toList(),
      };

  factory StudyPlan.fromJson(Map<String, dynamic> j) => StudyPlan(
        id: j['id'] as String,

class TaskItem {
  String id;
  String title;
  String notes;
  String dose;
  TaskKind kind;
  TaskTiming timing;
  RepeatMode repeat;
  List<int> weekdays; // 1 = Monday ... 7 = Sunday
  int intervalDays;
  DateTime startDate;
  DateTime? endDate;
  List<TaskTime> times;
  List<int> remindBefore; // minutes before due time
  int followUpMinutes; // "did you do it?" repeat gap after due time
  int nagCount; // how many follow-up reminders after due time
  int snoozeMinutes;
  bool alarmStyle; // loud full-screen alarm
  bool notify;
  bool archived;
  String? planId;
  int createdAt;

  TaskItem({
    required this.id,
    required this.title,
    this.notes = '',
    this.dose = '',
    required this.kind,
    this.timing = TaskTiming.atTime,
    this.repeat = RepeatMode.daily,
    List<int>? weekdays,
    this.intervalDays = 1,
    required this.startDate,
    this.endDate,
    required this.times,
    List<int>? remindBefore,
    this.followUpMinutes = 15,
    this.nagCount = 0,
    this.snoozeMinutes = 10,
    this.alarmStyle = false,
    this.notify = true,
    this.archived = false,
    this.planId,
    int? createdAt,
  })  : weekdays = weekdays ?? [1, 2, 3, 4, 5, 6, 7],
        remindBefore = remindBefore ?? [],
        createdAt = createdAt ?? DateTime.now().millisecondsSinceEpoch;

  bool get hasStreak => repeat != RepeatMode.once;

  bool isScheduledOn(DateTime d) {
    final day = dayOnly(d);
    final s = dayOnly(startDate);
    if (day.isBefore(s)) return false;
    if (endDate != null && day.isAfter(dayOnly(endDate!))) return false;
    return switch (repeat) {
      RepeatMode.once => day == s,
      RepeatMode.daily => true,
      RepeatMode.weekdays => weekdays.contains(day.weekday),
      RepeatMode.interval =>
        ((day.difference(s).inHours / 24).round() % (intervalDays < 1 ? 1 : intervalDays)) == 0,
    };
  }

  String get summary {
    const names = ['', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun'];
    final when = times.map((t) => fmtHm(t.hour, t.minute)).join(', ');
    if (kind == TaskKind.task) {
      final verb = timing == TaskTiming.completeBy ? 'Complete by' : 'At';
      return '$verb $when, ${fmtDate(startDate)}';
    }
    final r = switch (repeat) {
      RepeatMode.once => 'Once, ${fmtDate(startDate)}',
      RepeatMode.daily => 'Daily',
      RepeatMode.weekdays =>
        weekdays.length == 7 ? 'Daily' : weekdays.map((d) => names[d]).join(' '),
      RepeatMode.interval => 'Every $intervalDays days',
    };
    final e = (endDate != null && repeat != RepeatMode.once) ? ', until ${fmtDate(endDate!)}' : '';
    return '$r  |  $when$e';
  }

  Map<String, dynamic> toJson() => {
        'id': id,
        'title': title,
        'notes': notes,
        'dose': dose,
        'kind': kind.index,
        'timing': timing.index,
        'repeat': repeat.index,
        'weekdays': weekdays,
        'interval': intervalDays,
        'start': dateKey(startDate),
        'end': endDate == null ? null : dateKey(endDate!),
        'times': times.map((t) => t.toJson()).toList(),
        'remind': remindBefore,
        'follow': followUpMinutes,
        'nag': nagCount,
        'snooze': snoozeMinutes,
        'alarm': alarmStyle,
        'notify': notify,
        'archived': archived,
        'plan': planId,
        'created': createdAt,
      };

  factory TaskItem.fromJson(Map<String, dynamic> j) => TaskItem(
        id: j['id'] as String,
        title: j['title'] as String,
        notes: (j['notes'] ?? '') as String,
        dose: (j['dose'] ?? '') as String,
        kind: TaskKind.values[(j['kind'] ?? 0) as int],
        timing: TaskTiming.values[(j['timing'] ?? 0) as int],
        repeat: RepeatMode.values[(j['repeat'] ?? 1) as int],
        weekdays: List<int>.from((j['weekdays'] ?? [1, 2, 3, 4, 5, 6, 7]) as List),
        intervalDays: (j['interval'] ?? 1) as int,
        startDate: parseKey(j['start'] as String),
        endDate: j['end'] == null ? null : parseKey(j['end'] as String),
        times: ((j['times'] ?? []) as List)
            .map((e) => TaskTime.fromJson(Map<String, dynamic>.from(e as Map)))
            .toList(),
        remindBefore: List<int>.from((j['remind'] ?? []) as List),
        followUpMinutes: (j['follow'] ?? 15) as int,
        nagCount: (j['nag'] ?? 0) as int,
        snoozeMinutes: (j['snooze'] ?? 10) as int,
        alarmStyle: (j['alarm'] ?? false) as bool,
        notify: (j['notify'] ?? true) as bool,
        archived: (j['archived'] ?? false) as bool,
        planId: j['plan'] as String?,
        createdAt: (j['created'] ?? 0) as int,
      );

class _Detail extends StatelessWidget {
  final TaskItem task;
  const _Detail({required this.task});

  @override
  Widget build(BuildContext context) {
    final st = context.watch<AppState>();
    final g = context.glass;
    final today = dayOnly(DateTime.now());
    final frozen = st.frozen[task.id] ?? <String>{};
    final info = task.hasStreak ? st.streakFor(task) : null;

    var scheduled = 0, done = 0;
    for (var i = 0; i < 30; i++) {
      final d = addDays(today, -i);
      if (!task.isScheduledOn(d)) continue;
      if (i == 0 && !st.dayDone(task, dateKey(d))) continue;
      scheduled++;
      if (st.dayDone(task, dateKey(d))) done++;
    }
    final rate = scheduled == 0 ? 0 : (done * 100 / scheduled).round();

    Widget stat(String v, String l, IconData i, Color c) => Expanded(
          child: GlassContainer(
            radius: 20,
            blur: 12,
            padding: const EdgeInsets.all(14),
            child: Column(children: [
              Icon(i, color: c),
              const SizedBox(height: 6),
              Text(v,
                  style: TextStyle(
                      fontSize: 22,
                      fontWeight: FontWeight.w800,
                      color: g.text)),
              Text(l, style: TextStyle(fontSize: 11.5, color: g.subText)),
            ]),
          ),
        );

    Color? cell(DateTime d) {
      if (!task.isScheduledOn(d)) return null;
      final dk = dateKey(d);
      if (st.dayDone(task, dk)) return AppTheme.green;
      if (frozen.contains(dk)) return AppTheme.cyan;
      if (d == today) return AppTheme.blue;
      if (d.isAfter(today)) return g.text.withValues(alpha: 0.12);
      return AppTheme.red;
    }

    return Column(children: [
      Row(children: [
        stat('${info?.current ?? 0}', 'current',
            Icons.local_fire_department_rounded, AppTheme.amber),
        const SizedBox(width: 10),
        stat('${info?.best ?? 0}', 'best', Icons.emoji_events_rounded,
            AppTheme.purple),
        const SizedBox(width: 10),
        stat('$rate%', 'last 30 days', Icons.insights_rounded, AppTheme.cyan),
      ]),
      const SizedBox(height: 14),
      GlassContainer(
        padding: const EdgeInsets.all(14),
        child: Column(crossAxisAlignment: CrossAxisAlignment.start, children: [
          const SectionLabel('Last 5 weeks (tap a day to correct it)'),
          GridView.count(
            crossAxisCount: 7,
            shrinkWrap: true,
            physics: const NeverScrollableScrollPhysics(),
            mainAxisSpacing: 6,
            crossAxisSpacing: 6,
            children: [
              for (var i = 34; i >= 0; i--)
                Builder(builder: (_) {
                  final d = addDays(today, -i);

                  return GestureDetector(
                    onTap: c == null || d.isAfter(today)
                        ? null
                        : () => _fix(context, st, d),

                      decoration: BoxDecoration(
                        color: c?.withValues(alpha: 0.75) ??

/// Wraps the app in a fingerprint / face / PIN lock when enabled in Settings.
class LockGate extends StatefulWidget {
  const LockGate({super.key});

  @override
  State<LockGate> createState() => _LockGateState();
}

class _LockGateState extends State<LockGate> with WidgetsBindingObserver {
  bool _locked = false;
  bool _busy = false;
  DateTime? _pausedAt;

  @override
  void initState() {
    super.initState();
    WidgetsBinding.instance.addObserver(this);
    _locked = context.read<AppState>().settings.appLock;
    if (_locked) WidgetsBinding.instance.addPostFrameCallback((_) => _unlock());
  }

  @override
  void dispose() {
    WidgetsBinding.instance.removeObserver(this);
    super.dispose();
  }

  @override
  void didChangeAppLifecycleState(AppLifecycleState s) {
    if (_busy) return;
    if (s == AppLifecycleState.paused) {
      _pausedAt = DateTime.now();
    } else if (s == AppLifecycleState.resumed) {
      final lockOn = context.read<AppState>().settings.appLock;
      final away = _pausedAt == null ? Duration.zero : DateTime.now().difference(_pausedAt!);
      if (lockOn && !_locked && away.inSeconds > 20) {
        setState(() => _locked = true);
        _unlock();
      }
    }
  }

  Future<void> _unlock() async {
    if (_busy) return;
    _busy = true;
    final ok = await AuthService.authenticate('Unlock ConsistIQ');
    _busy = false;
    if (ok && mounted) setState(() => _locked = false);
  }

  @override
  Widget build(BuildContext context) {
    final g = context.glass;
    return Stack(children: [
      const RootShell(),
      if (_locked)
        Positioned.fill(
          child: Scaffold(
            body: GlassBackground(
              child: Center(
                child: Column(mainAxisSize: MainAxisSize.min, children: [
                  GlassContainer(
                    width: 110,
                    height: 110,
                    radius: 34,
                    glowColor: AppTheme.blue,
                    glowOpacity: 0.4,
                    child: Icon(Icons.fingerprint_rounded, size: 60, color: g.text),
                  ),
                  const SizedBox(height: 24),
                  Text('ConsistIQ is locked',
                      style: TextStyle(fontSize: 20, fontWeight: FontWeight.w800, color: g.text)),
                  const SizedBox(height: 20),
                  GlassButton(label: 'Unlock', icon: Icons.lock_open_rounded, filled: true, onTap: _unlock),
                ]),
              ),
            ),
          ),
        ),
    ]);
  }
}

  final _days = TextEditingController();
  final _min = TextEditingController(text: '90');
  TimeOfDay _time = const TimeOfDay(hour: 20, minute: 0);
  bool _loading = false;
  SplitResult? _result;

  @override
  void dispose() {
    _title.dispose();
    _text.dispose();
    _days.dispose();
    _min.dispose();
    super.dispose();
  }

  Future<void> _generate() async {
    final text = _text.text.trim();
    if (text.length < 3) return toast(context, 'Paste your syllabus first');
    final s = context.read<AppState>().settings;
    setState(() => _loading = true);
    final r = await SyllabusSplitter.split(
      text: text,
      days: int.tryParse(_days.text.trim()),
      minutesPerDay: int.tryParse(_min.text.trim()) ?? 90,
      apiKey: s.geminiKey,
      model: s.geminiModel,
    );
    if (!mounted) return;
    setState(() {
      _loading = false;
      _result = r;
    });
  }

  Future<void> _create() async {
    final r = _result;
    if (r == null || r.days.isEmpty) return;
    final title = _title.text.trim().isEmpty ? 'Study plan' : _title.text.trim();
    final planId = const Uuid().v4();
    final taskId = const Uuid().v4();
    final start = dayOnly(DateTime.now());
    final plan = StudyPlan(
      id: planId,
      title: title,
      taskId: taskId,
      source: r.source,
      createdAt: DateTime.now().millisecondsSinceEpoch,
      hour: _time.hour,
      minute: _time.minute,
      days: [
        for (var i = 0; i < r.days.length; i++)
          StudyDay(
            title: r.days[i].title,
            dateKey: dateKey(addDays(start, i)),
            points: r.days[i].points.map((p) => StudyPoint(p)).toList(),
          ),
      ],
    );

      id: taskId,
      title: title,
      kind: TaskKind.study,

      endDate: addDays(start, r.days.length - 1),
      times: [TaskTime('Study', _time.hour, _time.minute)],
      remindBefore: [30],
      followUpMinutes: 30,
      nagCount: 1,
      snoozeMinutes: 15,
      planId: planId,
    );
    await context.read<AppState>().addPlan(plan, task);
    if (mounted) Navigator.of(context).pop();
  }

/// Reads events from the phone's calendars (this includes Google Calendar
/// when your Google account is synced) and turns picked ones into reminders.
class CalendarImportScreen extends StatefulWidget {
  const CalendarImportScreen({super.key});

  @override
  State<CalendarImportScreen> createState() => _CalendarImportScreenState();
}

class _CalendarImportScreenState extends State<CalendarImportScreen> {
  final DeviceCalendarPlugin _plugin = DeviceCalendarPlugin();
  List<Calendar> _cals = [];
  Calendar? _sel;
  List<Event> _events = [];
  final Set<String> _picked = {};
  String? _msg;
  bool _busy = true;

  @override
  void initState() {
    super.initState();
    _init();
  }

  String _id(Event e) =>
      e.eventId ?? '${e.title}-${e.start?.millisecondsSinceEpoch}';

  DateTime _start(Event e) =>
      DateTime.fromMillisecondsSinceEpoch(e.start!.millisecondsSinceEpoch);

  Future<void> _init() async {
    try {
      var p = await _plugin.hasPermissions();
      if (p.data != true) p = await _plugin.requestPermissions();
      if (p.data != true) {
        if (mounted) {
          setState(() {
            _msg =
                'Calendar permission was denied. Allow it in phone settings.';
            _busy = false;
          });
        }
        return;
      }
      final r = await _plugin.retrieveCalendars();
      final list = (r.data ?? []).cast<Calendar>().toList();
      if (!mounted) return;
      setState(() {
        _cals = list;
        _busy = false;
        if (list.isEmpty) _msg = 'No calendars found on this phone.';
      });
      if (list.isNotEmpty) await _load(list.first);
    } catch (e) {
      if (mounted) {
        setState(() {
          _msg = 'Could not read calendar: $e';
          _busy = false;
        });
      }
    }
  }

  Future<void> _load(Calendar c) async {
    setState(() {
      _sel = c;
      _busy = true;
      _picked.clear();
    });
    try {
      final now = DateTime.now();
      final r = await _plugin.retrieveEvents(
        c.id,
        RetrieveEventsParams(
            startDate: now, endDate: now.add(const Duration(days: 21))),
      );
      final ev = (r.data ?? [])
          .cast<Event>()
          .where((e) =>
              e.start != null && e.title != null && _start(e).isAfter(now))
          .toList()
        ..sort((a, b) => _start(a).compareTo(_start(b)));
      if (mounted) {
        setState(() {
          _events = ev;
          _busy = false;
        });
      }
    } catch (e) {
      if (mounted) {
        setState(() {
          _msg = 'Could not load events: $e';
          _busy = false;
        });
      }
    }
  }

  Future<void> _import() async {
    final st = context.read<AppState>();
    var n = 0;
    for (final e in _events) {
      if (!_picked.contains(_id(e))) continue;
      final s = _start(e);
      final allDay = e.allDay == true;
      await st.saveTask(TaskItem(
        id: const Uuid().v4(),
        title: e.title ?? 'Event',
        notes: 'Imported from calendar',
        kind: TaskKind.meeting,
        repeat: TaskItem.fromJson({'repeat': 0}).repeat,
        startDate: dayOnly(s),
        times: [TaskTime('Event', allDay ? 9 : s.hour, allDay ? 0 : s.minute)],
        remindBefore: [60, 15],
        nagCount: 0,
      ));
      n++;
    }
    if (mounted) {
      toast(context, 'Added $n reminder${n == 1 ? '' : 's'}');
      Navigator.of(context).pop();
    }
  }

  @override
  Widget build(BuildContext context) {
    final g = context.glass;
    return PageShell(
      child: ListView(
        padding: const EdgeInsets.fromLTRB(20, 16, 20, 40),
        children: [
          const SubHeader(title: 'Calendar import'),
          if (_msg != null)
            GlassContainer(
                padding: const EdgeInsets.all(18),
                child: Text(_msg!, style: TextStyle(color: g.text))),
          if (_cals.isNotEmpty) ...[
            SingleChildScrollView(
              scrollDirection: Axis.horizontal,
              child: Row(children: [
                for (final c in _cals) ...[
                  GlassChip(
                      label: c.name ?? 'Calendar',
                      selected: _sel?.id == c.id,
                      onTap: () => _load(c)),
                  const SizedBox(width: 8),
                ],
              ]),
            ),
            const SizedBox(height: 16),
          ],
          if (_busy)
            const Center(
                child: Padding(
                    padding: EdgeInsets.all(24),
                    child: CircularProgressIndicator())),
          if (!_busy && _events.isEmpty && _msg == null && _sel != null)
            GlassContainer(
              padding: const EdgeInsets.all(18),
              child: Text('No upcoming events in the next 3 weeks.',
                  style: TextStyle(color: g.subText)),
            ),
          for (final e in _events)
            Padding(
              padding: const EdgeInsets.only(bottom: 10),
              child: GlassContainer(
                radius: 18,
                blur: 12,
                child: CheckboxListTile(
                  value: _picked.contains(_id(e)),
                  onChanged: (v) => setState(() =>
                      v == true ? _picked.add(_id(e)) : _picked.remove(_id(e))),
                  title: Text(e.title ?? '',
                      style: TextStyle(
                          color: g.text, fontWeight: FontWeight.w700)),
                  subtitle: Text(fmtDateTime(_start(e)),
                      style: TextStyle(color: g.subText)),
                ),
              ),
            ),
          if (_picked.isNotEmpty) ...[
            const SizedBox(height: 8),
            GlassButton(
              label: 'Add ${_picked.length} as reminders',
              icon: Icons.check_rounded,
              filled: true,
              color: AppTheme.purple,
              onTap: _import,
            ),
          ],
          const SizedBox(height: 16),
          Text(
            'Gmail: direct inbox import needs a Google restricted-scope approval, so it is not included. '
            'Events Gmail adds to Google Calendar (flights, bookings) appear here automatically.',
            style: TextStyle(fontSize: 12, color: g.subText),
          ),
        ],
      ),
    );
  }
}

class _GmailImportScreenState extends State<GmailImportScreen> {
  GoogleSignInAccount? _account;
  List<GmailMessage> _messages = [];
  bool _loading = false;
  String? _error;

  Future<void> _signIn() async {
    setState(() {
      _loading = true;
      _error = null;
    });
    try {
      final account = await GmailService.signIn();
      if (!mounted) return;
      setState(() => _account = account);
    } catch (e) {
      if (mounted) setState(() => _error = e.toString());
    } finally {
      if (mounted) setState(() => _loading = false);
    }
  }

  Future<void> _readInbox() async {
    final account = _account;
    if (account == null) return;
    final allow = await showDialog<bool>(
      context: context,
      builder: (context) => AlertDialog(
        title: const Text('Allow Gmail metadata access?'),
        content: const Text(
          'ConsistIQ will read the subject, sender, and short preview of up to 20 '
          'recent inbox emails so you can choose which ones to turn into tasks. '
          'Google grants metadata-only access: the app does not request message '
          'body access and cannot send, delete, or change emails. Only email '
          'details you choose to save are copied into tasks on this device. '
          'You can cancel or revoke access in your Google Account settings.',
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(context, false),
            child: const Text('Cancel'),
          ),
          FilledButton(
            onPressed: () => Navigator.pop(context, true),
            child: const Text('Continue'),
          ),
        ],
      ),
    );
    if (allow != true || !mounted) return;
    setState(() {
      _loading = true;
      _error = null;
    });
    try {
      final messages = await GmailService.readRecentInbox(account);
      if (mounted) setState(() => _messages = messages);
    } catch (e) {
      if (mounted) setState(() => _error = e.toString());
    } finally {
      if (mounted) setState(() => _loading = false);
    }
  }

  Future<void> _signOut() async {
    setState(() {
      _loading = true;
      _error = null;
    });
    try {
      await GmailService.signOut();
      if (mounted) {
        setState(() {
          _account = null;
          _messages = [];
        });
      }
    } catch (e) {
      if (mounted) setState(() => _error = e.toString());
    } finally {
      if (mounted) setState(() => _loading = false);
    }
  }

    final o = await StudyModeService.hasOverlay();
    final d = await StudyModeService.hasDnd();
    if (!mounted) return;

      _active = s.active;

class TaskEditorScreen extends StatefulWidget {
  final TaskItem? task;

    super.key,

  });

class _TaskEditorScreenState extends State<TaskEditorScreen> {
  final _title = TextEditingController();
  final _dose = TextEditingController();
  final _notes = TextEditingController();

  late TaskKind kind;
  TaskTiming timing = TaskTiming.atTime;
  RepeatMode repeat = RepeatMode.daily;
  Set<int> weekdays = {1, 2, 3, 4, 5, 6, 7};
  int interval = 2;
  late DateTime start;
  DateTime? end;
  List<TaskTime> times = [];
  Set<int> remind = {};
  int follow = 15, nag = 0, snooze = 10;
  bool alarm = false, notify = true;

  bool get editing => widget.task != null;

  @override
  void initState() {
    super.initState();
    start = dayOnly(DateTime.now());
    final t = widget.task;
    if (t != null) {
      kind = t.kind;
      timing = t.timing;
      _title.text = t.title;
      _dose.text = t.dose;
      _notes.text = t.notes;
      repeat = t.repeat;
      weekdays = t.weekdays.toSet();
      interval = t.intervalDays;
      start = dayOnly(t.startDate);
      end = t.endDate;
      times = t.times.map((e) => TaskTime(e.label, e.hour, e.minute)).toList();
      remind = t.remindBefore.toSet();
      follow = t.followUpMinutes;
      nag = t.nagCount;
      snooze = t.snoozeMinutes;
      alarm = t.alarmStyle;
      notify = t.notify;
    } else {
      _defaults(widget.kind ?? TaskKind.habit);
      _applyUserDefaults();
      _title.text = widget.initialTitle;
      _notes.text = widget.initialNotes;
    }
  }

  @override
  void dispose() {
    _title.dispose();
    _dose.dispose();
    _notes.dispose();
    super.dispose();
  }

  void _defaults(TaskKind k) {
    kind = k;
    timing = TaskTiming.atTime;
    final today = dayOnly(DateTime.now());
    start = today;
    switch (k) {
      case TaskKind.medicine:
        repeat = RepeatMode.daily;
        times = [
          TaskTime('Morning (after breakfast)', 8, 30),
          TaskTime('Afternoon (after lunch)', 14, 0),
          TaskTime('Night (after dinner)', 21, 0),
        ];
        remind = {};
        nag = 2;
        follow = 15;
        alarm = true;
        end = addDays(today, 14); // 15 days
        break;
      case TaskKind.deadline:
        repeat = RepeatMode.once;
        times = [TaskTime('Deadline', 23, 59)];
        remind = {60};
        nag = 0;
        follow = 15;
        alarm = false;
        end = null;
        break;
      case TaskKind.meeting:
        repeat = RepeatMode.once;
        final n = DateTime.now().add(const Duration(hours: 1));
        times = [TaskTime('Meeting', n.hour, 0)];
        remind = {60, 15};
        nag = 3;
        follow = 10;
        alarm = true;
        end = null;
        break;
      case TaskKind.task:
        repeat = RepeatMode.once;
        final n = DateTime.now().add(const Duration(hours: 1));
        start = dayOnly(n);
        times = [TaskTime('At time', n.hour, n.minute)];
        remind = {15};
        nag = 0;
        follow = 15;
        alarm = false;
        end = null;
        break;
      default:
        repeat = RepeatMode.daily;
        times = [TaskTime('Daily', 20, 0)];
        remind = {30};
        nag = 1;
        follow = 30;
        alarm = false;
        end = null;
    }
  }

  void _applyUserDefaults() {
    final settings = context.read<AppState>().settings;
    snooze = settings.defaultSnoozeMinutes;
    if (kind == TaskKind.task || kind == TaskKind.habit) {
      remind = settings.defaultReminderMinutes == 0
          ? {}
          : {settings.defaultReminderMinutes};
      alarm = settings.defaultAlarmStyle;
    }
  }

  Future<void> _pickTime(TaskTime t) async {
    final tp = await showTimePicker(
        context: context,
        initialTime: TimeOfDay(hour: t.hour, minute: t.minute));
    if (tp != null) {
      setState(() {
        t.hour = tp.hour;
        t.minute = tp.minute;
      });
    }
  }

  Future<void> _pickStart() async {
    final d = await showDatePicker(
      context: context,
      initialDate: start,
      firstDate: addDays(dayOnly(DateTime.now()), -30),
      lastDate: addDays(dayOnly(DateTime.now()), 730),
    );
    if (d == null) return;
    setState(() {
      if (end != null) {
        final len = end!.difference(start).inDays;
        end = addDays(d, len);
      }
      start = dayOnly(d);
    });
  }

  Future<void> _pickEnd() async {
    final d = await showDatePicker(
      context: context,
      initialDate: end ?? addDays(start, 6),
      firstDate: start,
      lastDate: addDays(start, 1500),
    );
    if (d != null) setState(() => end = dayOnly(d));
  }

  int? get _durationDays =>
      end == null ? null : end!.difference(start).inDays + 1;

  Future<void> _save() async {
    final title = _title.text.trim();
    if (title.isEmpty) return toast(context, 'Enter a title');
    if (times.isEmpty) return toast(context, 'Add at least one time');
    if (repeat == RepeatMode.weekdays && weekdays.isEmpty) {
      return toast(context, 'Pick at least one weekday');
    }
    final old = widget.task;
    final item = TaskItem(
      id: old?.id ?? const Uuid().v4(),
      title: title,
      notes: _notes.text.trim(),
      dose: _dose.text.trim(),
      kind: kind,
      timing: timing,
      repeat: repeat,
      weekdays: (weekdays.toList()..sort()),
      intervalDays: interval,
      startDate: start,
      endDate: repeat == RepeatMode.once ? null : end,
      times: times,
      remindBefore: (remind.toList()..sort()),
      followUpMinutes: follow,
      nagCount: nag,
      snoozeMinutes: snooze,
      alarmStyle: alarm,
      notify: notify,
      planId: old?.planId,
      createdAt: old?.createdAt,
    );
    await context.read<AppState>().saveTask(item);
    if (mounted) Navigator.of(context).pop();
  }

class GmailMessage {
  final String id;
  final String subject;
  final String sender;
  final String snippet;

  const GmailMessage({
    required this.id,
    required this.subject,
    required this.sender,
    required this.snippet,
  });

class GmailService {
  GmailService._();

  static const serverClientId =
      String.fromEnvironment('GOOGLE_SERVER_CLIENT_ID');
  static const _gmailMetadataScope =
      'https://www.googleapis.com/auth/gmail.metadata';
  static Future<void>? _initialization;

  static bool get isConfigured => serverClientId.isNotEmpty;

  static Future<void> _initialize() {
    if (!isConfigured) {
      throw StateError(
        'Google sign-in is not configured. Set GOOGLE_SERVER_CLIENT_ID when building the app.',
      );
    }
    return _initialization ??= GoogleSignIn.instance.initialize(
      serverClientId: serverClientId,
    );
  }

  static Future<GoogleSignInAccount> signIn() async {
    await _initialize();
    return GoogleSignIn.instance.authenticate();
  }

  static Future<void> signOut() async {
    await _initialize();
    await GoogleSignIn.instance.signOut();
  }

  static Future<List<GmailMessage>> readRecentInbox(
      GoogleSignInAccount account) async {
    final authorization =
        await account.authorizationClient.authorizeScopes(
      const [_gmailMetadataScope],
    );
    final headers = {
      'Authorization': 'Bearer ${authorization.accessToken}',
      'Accept': 'application/json',
    };
    final listResponse = await http
        .get(
          Uri.https(
            'gmail.googleapis.com',
            '/gmail/v1/users/me/messages',
            {'maxResults': '20', 'labelIds': 'INBOX'},
          ),
          headers: headers,
        )
        .timeout(const Duration(seconds: 20));
    if (listResponse.statusCode != 200) {
      throw StateError(
        'Gmail could not list recent messages (HTTP ${listResponse.statusCode}).',
      );
    }
    final listed = jsonDecode(listResponse.body) as Map<String, dynamic>;
    final items = (listed['messages'] as List<dynamic>? ?? const [])
        .whereType<Map<String, dynamic>>()
        .take(20)
        .toList();

    return Future.wait(items.map((item) async {
      final id = item['id'] as String;
      final uri = Uri.parse(
        'https://gmail.googleapis.com/gmail/v1/users/me/messages/$id'
        '?format=metadata&metadataHeaders=Subject&metadataHeaders=From',
      );
      final response = await http
          .get(uri, headers: headers)
          .timeout(const Duration(seconds: 20));
      if (response.statusCode != 200) {
        throw StateError(
          'Gmail could not read a selected message (HTTP ${response.statusCode}).',
        );
      }
      final message = jsonDecode(response.body) as Map<String, dynamic>;
      final payload = message['payload'] as Map<String, dynamic>? ?? const {};
      final messageHeaders =
          payload['headers'] as List<dynamic>? ?? const [];
      String header(String name) {
        for (final raw in messageHeaders) {
          if (raw is Map<String, dynamic> &&
              (raw['name'] as String?)?.toLowerCase() == name.toLowerCase()) {
            return (raw['value'] as String?) ?? '';
          }
        }
        return '';
      }

      return GmailMessage(
        id: id,
        subject: header('Subject').trim().isEmpty
            ? '(No subject)'
            : header('Subject').trim(),
        sender: header('From'),
        snippet: (message['snippet'] as String?) ?? '',
      );
    }));
  }
}

class HomeWidgetService {
  static const _quickAdd = 'com.example.app.widgets.QuickAddWidgetProvider';
  static const _today = 'com.example.app.widgets.TodayWidgetProvider';
  static const _progress = 'com.example.app.widgets.ProgressWidgetProvider';

  static Future<void> sync(
      List<TaskItem> tasks, Map<String, Completion> completions) async {
    if (!Platform.isAndroid) return;

    final today = dayOnly(DateTime.now());
    final dayKey = dateKey(today);
    final occurrences = <({TaskItem task, int slot, bool done})>[];
    for (final task in tasks) {
      if (task.archived || !task.isScheduledOn(today)) continue;
      for (var slot = 0; slot < task.times.length; slot++) {
        occurrences.add((
          task: task,
          slot: slot,
          done: completions[occKey(task.id, dayKey, slot)]?.status == 'done',
        ));
      }
    }

    final done = occurrences.where((item) => item.done).length;
    final remaining = occurrences
        .where((item) => !item.done)
        .take(4)
        .map((item) => '${item.task.kind.label}: ${item.task.title}')
        .toList();

    await _save('today_tasks', jsonEncode(remaining));
    await _save('today_done', '$done');
    await _save('today_total', '${occurrences.length}');

    for (final provider in [_quickAdd, _today, _progress]) {
      await HomeWidget.updateWidget(qualifiedAndroidName: provider);
    }
  }

  static Future<void> _save(String key, String value) async {
    final saved = await HomeWidget.saveWidgetData<String>(key, value);
    if (saved != true) {
      throw StateError('Could not save $key for Android home-screen widgets.');
    }
  }

class ReminderPayload {
  final String taskId;
  final String dateKey;
  final int slot;
  const ReminderPayload(this.taskId, this.dateKey, this.slot);

  String encode() => jsonEncode({'t': taskId, 'd': dateKey, 's': slot});

  static ReminderPayload? tryParse(String? s) {
    if (s == null || s.isEmpty) return null;
    try {
      final m = jsonDecode(s) as Map<String, dynamic>;
      return ReminderPayload(m['t'] as String, m['d'] as String, m['s'] as int);
    } catch (_) {
      return null;
    }
  }
}

/// Runs in a background isolate when a notification action button
/// (Taken / Done / Snooze) is pressed while the app is closed.
@pragma('vm:entry-point')
void notificationTapBackground(NotificationResponse r) async {
  DartPluginRegistrant.ensureInitialized();
  final p = ReminderPayload.tryParse(r.payload);
  final a = r.actionId;
  if (p == null || a == null || a.isEmpty || a == 'later') return;
  await NotificationService.instance.init(background: true);
  final s = AppState(background: true);
  await s.load();
  await s.handleAction(p, a);
  try {
    await HomeWidgetService.sync(s.tasks, s.comps);
  } catch (e) {
    debugPrint('Home widget refresh after notification action failed: $e');
  }
}

  /// Rebuilds every upcoming reminder from scratch. Completed occurrences get
  /// no reminders; active snoozes are preserved.
  Future<void> rescheduleAll(
    List<TaskItem> tasks,
    Map<String, Completion> comps,
    Map<String, int> snoozes,
    AppSettings settings,
  ) async {
    await _plugin.cancelAll();
    if (!settings.notificationsOn) return;
    await _ensureAlarmChannel(settings);
    final exact = (await _android?.canScheduleExactNotifications()) ?? false;
    final mode = exact
        ? AndroidScheduleMode.exactAllowWhileIdle
        : AndroidScheduleMode.inexactAllowWhileIdle;

    final now = DateTime.now();
    final today = dayOnly(now);
    final list = <_Planned>[];

    for (final t in tasks) {
      if (t.archived || !t.notify) continue;
      for (var off = -1; off < windowDays; off++) {
        final day = addDays(today, off);
        if (!t.isScheduledOn(day)) continue;
        final dk = dateKey(day);
        for (var i = 0; i < t.times.length; i++) {
          final k = occKey(t.id, dk, i);
          if (comps.containsKey(k)) continue; // done or skipped: no reminders
          final tt = t.times[i];
          final due =
              DateTime(day.year, day.month, day.day, tt.hour, tt.minute);
          final payload = ReminderPayload(t.id, dk, i);
          final sn = snoozes[k];
          if (sn != null) {
            list.add(_Planned(_nid(t.id, dk, i, 30),
                DateTime.fromMillisecondsSinceEpoch(sn), t, i, 'due', payload));
          }
          if (off < 0) continue; // yesterday: only keep snoozes
          for (var j = 0; j < t.remindBefore.length && j < 5; j++) {
            final m = t.remindBefore[j];
            list.add(_Planned(_nid(t.id, dk, i, j),
                due.subtract(Duration(minutes: m)), t, i, 'pre', payload,
                mins: m));
          }
          list.add(_Planned(_nid(t.id, dk, i, 10), due, t, i, 'due', payload));
          for (var n = 1; n <= t.nagCount && n <= 5; n++) {
            list.add(_Planned(
                _nid(t.id, dk, i, 20 + n),
                due.add(Duration(minutes: t.followUpMinutes * n)),
                t,
                i,
                'nag',
                payload));
          }
        }
      }
    }

    final limit = now.add(const Duration(seconds: 3));
    list.removeWhere((p) => !p.when.isAfter(limit));
    list.sort((a, b) => a.when.compareTo(b.when));

    for (final p in list.take(maxScheduled)) {
      final text = _text(p);
      try {
        await _plugin.zonedSchedule(
          p.id,
          text.$1,
          text.$2,
          tz.TZDateTime.from(p.when, tz.local),
          _details(p.task, p.phase, text.$2, settings),
          androidScheduleMode: mode,
          uiLocalNotificationDateInterpretation:
              UILocalNotificationDateInterpretation.absoluteTime,
          payload: p.payload.encode(),
        );
      } catch (e) {
        debugPrint('schedule failed: $e');
      }
    }
  }

  (String, String) _text(_Planned p) {
    final t = p.task;
    final tt = t.times[p.slot];
    final at = fmtHm(tt.hour, tt.minute);
    final med = t.kind == TaskKind.medicine;
    switch (p.phase) {
      case 'pre':
        final m = p.mins;
        final left = m >= 60 && m % 60 == 0 ? '${m ~/ 60} hr' : '$m min';
        return (
          med ? 'Medicine soon: ${t.title}' : 'Coming up: ${t.title}',
          'Due at $at, $left to go.${t.dose.isNotEmpty ? ' ${t.dose}.' : ''}'
        );
      case 'nag':
        return (
          'Still pending: ${t.title}',
          med
              ? 'Have you taken ${t.title}? Tap Taken, Snooze, or Not yet.'
              : 'Not marked done yet. Your streak is at risk!'
        );
      default:
        final title = switch (t.kind) {
          TaskKind.medicine => 'Time for ${t.title}',
          TaskKind.deadline => 'Deadline now: ${t.title}',
          TaskKind.meeting => 'Meeting: ${t.title}',
          TaskKind.study => 'Study time: ${t.title}',
          TaskKind.task => t.timing == TaskTiming.completeBy
              ? 'Complete by: ${t.title}'
              : 'Task reminder: ${t.title}',
          TaskKind.habit => 'Time for: ${t.title}',
        };
        final lab =
            tt.label.isNotEmpty && t.times.length > 1 ? '${tt.label}. ' : '';
        final dose = t.dose.isNotEmpty ? '${t.dose}. ' : '';
        return (
          title,
          med
              ? '$dose${lab}Have you taken it?'
              : 'Tap Done when finished, or snooze.'
        );
    }
  }

  String _alarmChannelId(AppSettings settings) =>
      'consistiq_alarm_${_hash(settings.alarmSoundUri)}_${settings.alarmVibration ? 1 : 0}';

            ? AndroidNotificationCategory.alarm

            ? AudioAttributesUsage.alarm

  }

/// Streak = consecutive scheduled days where every time-slot was marked done.
/// - Today still pending never breaks the streak.
/// - A "frozen" day (protected by a streak freeze) neither breaks nor adds.
StreakInfo computeStreak(
  TaskItem t,
  Map<String, Completion> comps,
  Set<String> frozen,
  DateTime asOf,
) {
  final today = dayOnly(asOf);
  var start = dayOnly(t.startDate);
  final earliest = addDays(today, -730);
  if (start.isBefore(earliest)) start = earliest;
  var run = 0;
  var best = 0;
  for (var d = start; !d.isAfter(today); d = addDays(d, 1)) {
    if (!t.isScheduledOn(d)) continue;
    final dk = dateKey(d);
    var done = t.times.isNotEmpty;
    for (var i = 0; i < t.times.length; i++) {
      if (comps[occKey(t.id, dk, i)]?.status != 'done') {
        done = false;
        break;
      }
    }
    if (done) {
      run++;
      if (run > best) best = run;
    } else if (d == today) {
      // still pending - keep the streak alive
    } else if (frozen.contains(dk)) {
      // protected by a freeze
    } else {
      run = 0;
    }
  }
  return StreakInfo(run, best);
}

class SplitDay {
  String title;
  List<String> points;
  SplitDay(this.title, this.points);
}

class SplitResult {
  final List<SplitDay> days;
  final String source;
  final String? note;
  SplitResult(this.days, this.source, {this.note});
}

class SyllabusSplitter {
  static Future<SplitResult> split({
    required String text,
    int? days,
    required int minutesPerDay,
    required String apiKey,
    required String model,
  }) async {
    String? note;
    if (apiKey.trim().isNotEmpty) {
      try {
        final r = await _gemini(text, days, minutesPerDay, apiKey.trim(), model.trim());
        if (r.isNotEmpty) return SplitResult(r, 'AI (Gemini)');
        note = 'AI returned nothing, used offline splitter.';
      } catch (e) {
        debugPrint('Gemini failed: $e');
        note = 'AI failed ($e). Used offline splitter.';
      }
    } else {
      note = 'No AI key set (Settings). Used offline splitter.';
    }
    return SplitResult(_local(text, days ?? 14), 'Offline splitter', note: note);
  }

  static Future<List<SplitDay>> _gemini(
      String text, int? days, int minutes, String key, String model) async {
    final uri = Uri.parse(
        'https://generativelanguage.googleapis.com/v1beta/models/$model:generateContent');
    final dayRule = days != null
        ? 'Use exactly $days days.'
        : 'Choose a sensible number of days (maximum 60).';
    final prompt = '''You are a study planner. Split the syllabus below into daily study tasks.
Rules: $dayRule Each day is about $minutes minutes of study. Order topics from basics to advanced. Split big topics across several days. Add a revision day now and then. Every day has a short title and 3 to 6 concrete checklist points (concepts to learn, exercises to practice).
Return ONLY JSON in this shape: {"days":[{"title":"...","points":["...","..."]}]}

SYLLABUS:
$text''';
    final res = await http
        .post(
          uri,
          headers: {'Content-Type': 'application/json', 'x-goog-api-key': key},
          body: jsonEncode({
            'contents': [
              {
                'parts': [
                  {'text': prompt}
                ]
              }
            ],
            'generationConfig': {'responseMimeType': 'application/json', 'temperature': 0.3},
          }),
        )
        .timeout(const Duration(seconds: 60));
    if (res.statusCode != 200) {
      final b = res.body;
      throw Exception('HTTP ${res.statusCode}: ${b.length > 160 ? b.substring(0, 160) : b}');
    }
    final data = jsonDecode(res.body);
    var out = data['candidates'][0]['content']['parts'][0]['text'] as String;
    out = out.replaceAll(RegExp(r'```json|```'), '').trim();
    final j = jsonDecode(out);
    final list = (j is List ? j : j['days']) as List;
    final result = <SplitDay>[];
    for (final e in list) {
      final m = Map<String, dynamic>.from(e as Map);
      final pts = ((m['points'] ?? []) as List).map((x) => x.toString()).toList();
      if (pts.isEmpty) continue;
      result.add(SplitDay((m['title'] ?? 'Study').toString(), pts));
    }
    return result;
  }

  /// Offline fallback: split by lines / commas / semicolons, group evenly.
  static List<SplitDay> _local(String text, int days) {
    final items = text
        .split(RegExp(r'[\n;,]+'))
        .map((s) => s.replaceAll(RegExp(r'^\s*([-*•]|\d+[.)])\s*'), '').trim())
        .where((s) => s.length > 1)
        .toList();
    if (items.isEmpty) return [];
    final n = days.clamp(1, items.length);
    final out = <SplitDay>[];
    for (var i = 0; i < n; i++) {
      final from = (items.length * i / n).floor();
      final to = (items.length * (i + 1) / n).floor();
      final chunk = items.sublist(from, to);
      if (chunk.isEmpty) continue;
      out.add(SplitDay(
        chunk.length == 1 ? chunk.first : '${chunk.first} and more',
        [
          ...chunk.map((c) => 'Learn: $c'),
          'Practice questions and write short notes',
        ],
      ));
    }
    return out;
  }
}

/// One due item: a task on a given day at a given time slot.
class Occ {
  final TaskItem task;
  final String dateKey;
  final int slot;
  final Completion? c;
  Occ(this.task, this.dateKey, this.slot, this.c);

  TaskTime get time => task.times[slot];
  DateTime get due {
    final d = parseKey(dateKey);
    return DateTime(d.year, d.month, d.day, time.hour, time.minute);
  }

  bool get done => c?.status == 'done';
  bool get skipped => c?.status == 'skipped';
  bool get overdue => c == null && due.isBefore(DateTime.now());
}

class AppState extends ChangeNotifier {
  AppState({this.background = false});
  final bool background;

  List<TaskItem> tasks = [];
  Map<String, Completion> comps = {};
  Map<String, Set<String>> frozen = {};
  Map<String, int> snoozes = {};
  List<StudyPlan> plans = [];
  AppSettings settings = AppSettings();
  int freezes = 1;
  String? settledThrough;
  int _rev = 0;

  // ---------------------------------------------------------------- storage
  List<T> _list<T>(String? s, T Function(Map<String, dynamic>) f) {
    if (s == null) return [];
    return (jsonDecode(s) as List).map((e) => f(Map<String, dynamic>.from(e as Map))).toList();
  }

  Future<void> load() async {
    final p = await SharedPreferences.getInstance();
    await p.reload();
    tasks = _list(p.getString('tasks'), TaskItem.fromJson);
    comps = {for (final c in _list(p.getString('comps'), Completion.fromJson)) c.key: c};
    plans = _list(p.getString('plans'), StudyPlan.fromJson);
    frozen = {};
    final fz = p.getString('frozen');
    if (fz != null) {
      (jsonDecode(fz) as Map<String, dynamic>).forEach((k, v) => frozen[k] = Set<String>.from(v as List));
    }
    snoozes = {};
    final sn = p.getString('snoozes');
    if (sn != null) {
      (jsonDecode(sn) as Map<String, dynamic>).forEach((k, v) => snoozes[k] = v as int);
    }
    final st = p.getString('settings');
    settings = st == null ? AppSettings() : AppSettings.fromJson(Map<String, dynamic>.from(jsonDecode(st) as Map));
    freezes = p.getInt('freezes') ?? 1;
    settledThrough = p.getString('settled');
    _rev = p.getInt('rev') ?? 0;
    if (_settle()) await _save();
    notifyListeners();
  }

  /// Called on app resume / periodically: picks up changes made by
  /// notification buttons (background isolate).
  Future<void> refreshFromDisk() async {
    final p = await SharedPreferences.getInstance();
    await p.reload();
    if ((p.getInt('rev') ?? 0) != _rev) await load();
  }

  Future<void> _save() async {
    final p = await SharedPreferences.getInstance();
    await p.reload();
    _rev = (p.getInt('rev') ?? 0) + 1;
    await p.setString('tasks', jsonEncode(tasks.map((e) => e.toJson()).toList()));
    await p.setString('comps', jsonEncode(comps.values.map((e) => e.toJson()).toList()));
    await p.setString('plans', jsonEncode(plans.map((e) => e.toJson()).toList()));
    await p.setString('frozen', jsonEncode(frozen.map((k, v) => MapEntry(k, v.toList()))));
    await p.setString('snoozes', jsonEncode(snoozes));
    await p.setString('settings', jsonEncode(settings.toJson()));
    await p.setInt('freezes', freezes);
    if (settledThrough != null) await p.setString('settled', settledThrough!);
    await p.setInt('rev', _rev);
  }

  bool _sched = false, _again = false;
  Future<void> reschedule() async {
    if (_sched) {
      _again = true;
      return;
    }
    _sched = true;
    try {
      do {
        _again = false;
        await NotificationService.instance.rescheduleAll(tasks, comps, snoozes, settings);
      } while (_again);
    } catch (e) {
      debugPrint('reschedule error: $e');
    } finally {
      _sched = false;
    }
  }

  Future<void> _commit({bool resched = true}) async {
    await _save();
    notifyListeners();
    if (!background) {
      unawaited(HomeWidgetService.sync(tasks, comps).catchError((e) {
        debugPrint('Home widget sync failed: $e');
      }));
    }
    if (resched) {
      if (background) {
        await reschedule();
      } else {
        unawaited(reschedule());
      }
    }
  }

  // --------------------------------------------------------------- lookups
  TaskItem? taskById(String? id) {
    if (id == null) return null;
    for (final t in tasks) {
      if (t.id == id) return t;
    }
    return null;
  }

  StudyPlan? planById(String? id) {
    if (id == null) return null;
    for (final p in plans) {
      if (p.id == id) return p;
    }
    return null;
  }

  List<Occ> occurrencesOn(DateTime day) {
    final dk = dateKey(day);
    final out = <Occ>[];
    for (final t in tasks) {
      if (t.archived || !t.isScheduledOn(day)) continue;
      for (var i = 0; i < t.times.length; i++) {
        out.add(Occ(t, dk, i, comps[occKey(t.id, dk, i)]));
      }
    }
    out.sort((a, b) => a.due.compareTo(b.due));
    return out;
  }

  bool dayDone(TaskItem t, String dk) {
    if (t.times.isEmpty) return false;
    for (var i = 0; i < t.times.length; i++) {
      if (comps[occKey(t.id, dk, i)]?.status != 'done') return false;
    }
    return true;
  }

  StreakInfo streakFor(TaskItem t) =>
      computeStreak(t, comps, frozen[t.id] ?? <String>{}, DateTime.now());

  // ------------------------------------------------------------ streak freeze
  bool _settle() {
    final today = dayOnly(DateTime.now());
    final yKey = dateKey(addDays(today, -1));
    final old = settledThrough;
    if (old == null) {
      settledThrough = yKey;
      return true;
    }
    var changed = false;
    var d = addDays(parseKey(old), 1);
    var guard = 0;
    while (!d.isAfter(addDays(today, -1)) && guard++ < 60) {
      final dk = dateKey(d);
      for (final t in tasks) {
        if (t.archived || !t.hasStreak || !t.isScheduledOn(d)) continue;
        if (dayDone(t, dk) || freezes <= 0) continue;
        final before = computeStreak(t, comps, frozen[t.id] ?? <String>{}, d).current;
        if (before > 0) {
          frozen.putIfAbsent(t.id, () => <String>{}).add(dk);
          freezes--;
          changed = true;
        }
      }
      d = addDays(d, 1);
    }
    snoozes.removeWhere((k, v) => v < DateTime.now().millisecondsSinceEpoch - 86400000);
    settledThrough = yKey;
    return changed || old != yKey;
  }

  // ------------------------------------------------------------------ tasks
  Future<void> saveTask(TaskItem t) async {
    final i = tasks.indexWhere((e) => e.id == t.id);
    if (i >= 0) {
      tasks[i] = t;
    } else {
      tasks.add(t);
    }
    await _commit();
  }

  void _removeTaskData(String id) {
    tasks.removeWhere((t) => t.id == id);
    comps.removeWhere((k, v) => v.taskId == id);
    frozen.remove(id);
    snoozes.removeWhere((k, v) => k.startsWith('$id|'));
  }

  Future<void> deleteTask(TaskItem t) async {
    _removeTaskData(t.id);
    if (t.planId != null) plans.removeWhere((p) => p.id == t.planId);
    await _commit();
  }

  // ------------------------------------------------------------ completions
  Future<void> complete(TaskItem t, String dk, int slot, {String source = 'tap'}) async {
    final wasDone = dayDone(t, dk);
    final k = occKey(t.id, dk, slot);
    comps[k] = Completion(
      taskId: t.id,
      dateKey: dk,
      slot: slot,
      status: 'done',
      ts: DateTime.now().millisecondsSinceEpoch,
      source: source,
    );
    snoozes.remove(k);
    if (!wasDone && dayDone(t, dk) && t.hasStreak) {
      final s = streakFor(t).current;
      if (s > 0 && s % 7 == 0 && freezes < 3) freezes++; // earn a freeze every 7 days
    }
    if (t.planId != null) planById(t.planId)?.markNextDone();
    await _commit();
  }

  Future<void> skip(TaskItem t, String dk, int slot) async {
    final k = occKey(t.id, dk, slot);
    comps[k] = Completion(
      taskId: t.id,
      dateKey: dk,
      slot: slot,
      status: 'skipped',
      ts: DateTime.now().millisecondsSinceEpoch,
      source: 'tap',
    );
    snoozes.remove(k);
    await _commit();
  }

  Future<void> undo(TaskItem t, String dk, int slot) async {
    comps.remove(occKey(t.id, dk, slot));
    await _commit();
  }

  /// Manual correction from the history screen (kept in the audit log).
  Future<void> setDayManual(TaskItem t, String dk, bool done) async {
    for (var i = 0; i < t.times.length; i++) {
      final k = occKey(t.id, dk, i);
      if (done) {
        comps[k] = Completion(
          taskId: t.id,
          dateKey: dk,
          slot: i,
          status: 'done',
          ts: DateTime.now().millisecondsSinceEpoch,
          source: 'manual',
        );
      } else {
        comps.remove(k);
      }
    }
    await _commit();
  }

  Future<void> snooze(TaskItem t, String dk, int slot, int minutes, {DateTime? until}) async {
    final when = until ?? DateTime.now().add(Duration(minutes: minutes));
    snoozes[occKey(t.id, dk, slot)] = when.millisecondsSinceEpoch;
    await _commit();
  }

  /// Notification action buttons (works from the background isolate too).
  Future<void> handleAction(ReminderPayload p, String action) async {
    final t = taskById(p.taskId);
    if (t == null) return;
    switch (action) {
      case 'done':
        await complete(t, p.dateKey, p.slot, source: 'notification');
        break;
      case 'snooze':
        await snooze(t, p.dateKey, p.slot, t.snoozeMinutes);
        break;
      case 'skip':
        await skip(t, p.dateKey, p.slot);
        break;
    }
  }

  // ------------------------------------------------------------ study plans
  Future<void> addPlan(StudyPlan plan, TaskItem task) async {
    plans.add(plan);
    tasks.add(task);
    await _commit();
  }

  Future<void> deletePlan(StudyPlan p) async {
    plans.removeWhere((e) => e.id == p.id);
    _removeTaskData(p.taskId);
    await _commit();
  }

  Future<void> togglePoint(StudyPoint pt) async {
    pt.done = !pt.done;
    await _commit(resched: false);
  }

  Future<void> completeStudyToday(StudyPlan p) async {
    final t = taskById(p.taskId);
    if (t == null) return;
    await complete(t, dateKey(DateTime.now()), 0);
  }

  /// Catch-up: re-date every unfinished day starting today. Finished work is untouched.
  Future<void> catchUp(StudyPlan p) async {
    var d = dayOnly(DateTime.now());
    var any = false;
    for (final day in p.days) {
      if (day.done) continue;
      day.dateKey = dateKey(d);
      d = addDays(d, 1);
      any = true;
    }
    final t = taskById(p.taskId);
    if (any && t != null) t.endDate = addDays(d, -1);
    await _commit();
  }

  // --------------------------------------------------------------- settings
  Future<void> updateSettings(void Function(AppSettings s) f, {bool resched = true}) async {
    f(settings);
    await _commit(resched: resched);
  }
}
